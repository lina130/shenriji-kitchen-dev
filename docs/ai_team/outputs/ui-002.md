## 3. 节点结构（可直接实施的 .tscn 结构）

### 3.1 场景实体基类：`scripts/gameplay/interaction_entity.gd`

```gdscript
class_name InteractionEntity
extends Area2D

@export var binding_id: String = ""
@export var entity_type: String = ""
@export var entity_id: String = ""
@export var asset_key: String = ""
@export var prompt_key: String = ""
@export var action_id: String = ""

var _sprite: Sprite2D
var _highlight: Sprite2D
var _prompt_label: Label
var _state: String = "normal"
var _hovered := false
var _busy := false

func _ready() -> void:
    input_pickable = true
    monitoring = false
    monitorable = false
    _build_visuals()
    mouse_entered.connect(_on_mouse_entered)
    mouse_exited.connect(_on_mouse_exited)
    input_event.connect(_on_input_event)
    _apply_state("normal")

func _build_visuals() -> void:
    _sprite = Sprite2D.new()
    _sprite.name = "Sprite"
    _sprite.texture = PresentationManager.get_ui_texture(asset_key, "normal")
    add_child(_sprite)

    _highlight = Sprite2D.new()
    _highlight.name = "Highlight"
    _highlight.texture = PresentationManager.get_ui_texture("interaction.highlight", "normal")
    _highlight.modulate = PresentationManager.get_token_color("interaction.highlight_color")
    _highlight.visible = false
    add_child(_highlight)

    _prompt_label = Label.new()
    _prompt_label.name = "PromptLabel"
    _prompt_label.text = PresentationManager.get_prompt_text(prompt_key)
    _prompt_label.visible = false
    _prompt_label.position = Vector2(0, -48)
    _prompt_label.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
    add_child(_prompt_label)

func _on_mouse_entered() -> void:
    if _busy:
        return
    _hovered = true
    _apply_state("hover")
    _highlight.visible = true
    _prompt_label.visible = true
    var tween := create_tween()
    tween.tween_property(self, "scale", Vector2.ONE * PresentationManager.get_token_float("interaction.hover_scale"), PresentationManager.get_token_float("interaction.hover_duration"))

func _on_mouse_exited() -> void:
    _hovered = false
    _apply_state("normal")
    _highlight.visible = false
    _prompt_label.visible = false
    var tween := create_tween()
    tween.tween_property(self, "scale", Vector2.ONE, PresentationManager.get_token_float("interaction.hover_duration"))

func _on_input_event(_viewport: Node, event: InputEvent, _shape_idx: int) -> void:
    if _busy:
        return
    if event is InputEventMouseButton and event.pressed and event.button_index == MOUSE_BUTTON_LEFT:
        _flash_click()
        InteractionRouter.dispatch(action_id, self)
        get_viewport().set_input_as_handled()

func _flash_click() -> void:
    var tween := create_tween()
    tween.tween_property(_sprite, "modulate", Color.WHITE * 1.4, PresentationManager.get_token_float("interaction.click_flash_duration") * 0.5)
    tween.tween_property(_sprite, "modulate", Color.WHITE, PresentationManager.get_token_float("interaction.click_flash_duration") * 0.5)

func set_busy(value: bool) -> void:
    _busy = value
    _apply_state("busy" if value else "normal")
    if value:
        var tween := create_tween().set_loops()
        tween.tween_property(_sprite, "modulate:a", 0.6, PresentationManager.get_token_float("interaction.busy_pulse_period") * 0.5)
        tween.tween_property(_sprite, "modulate:a", 1.0, PresentationManager.get_token_float("interaction.busy_pulse_period") * 0.5)

func _apply_state(state: String) -> void:
    _state = state
    var tex := PresentationManager.get_ui_texture(asset_key, state)
    if tex != null:
        _sprite.texture = tex
```

## 4. 输入流程（完整状态机）

### 4.1 点击工位 → 烹饪

```
[玩家点击 StationWok1]
  → InteractionEntity._on_input_event
  → InteractionRouter.dispatch("cook_at_station", {station_id: "wok_1"})
  → InteractionRouter 检查:
      - 工位是否 busy? → 是则 NoticeArbiter.push("工位正忙", "hint") 并 return
      - 玩家是否手持食材? → 否则 NoticeArbiter.push("先去食材架取食材", "hint") 并 return
  → 打开 RecipeTray，定位到工位屏幕坐标
  → 玩家点击 RecipeCard
  → InteractionRouter.dispatch("cook_at_station", {station_id, recipe_id})
  → KitchenManager.start_cooking(station_id, recipe_id)
  → StationWok1.set_busy(true)
  → RecipeTray.visible = false
  → 烹饪完成后 KitchenManager.changed 信号 → StationWok1.set_busy(false)
```

### 4.2 点击点餐台 → 查看订单

```
[玩家点击 CounterMain]
  → InteractionRouter.dispatch("pick_order", {counter_id: "counter_main"})
  → 打开 OrderTicketPanel（Control，非 PopupMenu）
  → 显示当前所有待处理订单卡
  → 玩家点击某张订单卡
  → InteractionRouter.dispatch("accept_order", {order_id})
  → 订单进入玩家"进行中"列表
  → OrderTicketPanel.visible = false
```

### 4.3 点击顾客 → 服务

```
[玩家点击 QueueSlot0 上的 CustomerActor]
  → InteractionRouter.dispatch("serve_customer", {customer_id})
  → 检查玩家是否持有该顾客订单
  → 是 → 交付 → CustomerActor 播放离开动画 → 队列前移
  → 否 → NoticeArbiter.push("这位顾客的订单还没做好", "hint")
```

### 4.4 点击食材架 → 取食材

```
[玩家点击 RackMain]
  → InteractionRouter.dispatch("open_rack", {rack_id: "rack_main"})
  → 打开 IngredientRackPanel
  → 玩家点击 IngredientSlot
  → InteractionRouter.dispatch("take_ingredient", {rack_id, item_id, count: 1})
  → InventoryManager.add_item(item_id, 1)
  → 暂存扣减 → 槽位刷新
```

### 4.5 点击货架 → 批发

```
[玩家点击 ShelfMain]
  → InteractionRouter.dispatch("buy_goods", {shelf_id: "shelf_main"})
  → 打开 GoodsShelfPanel（Control）
  → 玩家点击商品卡 → 数量选择用 +/- 按钮（非下拉）
  → 确认 → BusinessManager.buy_goods(goods_id, quantity)
```

### 4.6 点击员工站位 → 分配

```
[玩家点击 StaffPostKitchen]
  → InteractionRouter.dispatch("assign_staff", {post_id: "post_kitchen"})
  → 打开 StaffAssignPanel（Control）
  → 显示所有可用员工头像卡
  → 玩家点击员工卡 → 分配到该站位
  → 站位实体 set_busy(true) + 显示员工头像
```

1. `world.gd` 已有 `_find_click_target(mouse_event.position)` + `direct_target.interact()` —— **场景点击管线已存在**，本方案复用它，不新建。

```

```

**与主方案的分歧**：主方案倾向给每个经营界面单独做一套场景 UI；本方案主张**统一手势语义**——所有经营实体共享同一套"左键=主操作 / 右键=次操作 / 长按=详情 / 拖拽=转移"手势，实体只声明自己支持哪些手势。

### 1.1 新增节点（唯一新增）

```gdscript
# scripts/gameplay/click_gesture_router.gd
class_name ClickGestureRouter
extends Node

# 挂在 WorldRoot 下，_ready 时注册到 "click_router" 组
# 职责：把原始鼠标事件翻译成 4 种语义手势，转发给命中的 Interactable

enum Gesture { TAP, SECONDARY, HOLD, DRAG_DROP }

signal gesture_dispatched(target: Node, gesture: int, payload: Dictionary)

const HOLD_THRESHOLD := 0.35   # 秒
const DRAG_THRESHOLD := 8.0    # 像素

var _press_pos := Vector2.ZERO
var _press_time := 0.0
var _press_target: Node = null
var _dragging := false
var _drag_payload: Dictionary = {}

func _unhandled_input(event: InputEvent) -> void:
    if event is InputEventMouseButton:
        var mb := event as InputEventMouseButton
        if mb.button_index == MOUSE_BUTTON_LEFT:
            if mb.pressed:
                _begin_press(mb.position)
            else:
                _end_press(mb.position)
        elif mb.button_index == MOUSE_BUTTON_RIGHT and mb.pressed:
            _dispatch_secondary(mb.position)
    elif event is InputEventMouseMotion and _press_target != null:
        var mm := event as InputEventMouseMotion
        if not _dragging and mm.position.distance_to(_press_pos) > DRAG_THRESHOLD:
            _dragging = true
            _press_target.on_drag_begin(_drag_payload)

func _begin_press(pos: Vector2) -> void:
    _press_pos = pos
    _press_time = Time.get_ticks_msec() / 1000.0
    _press_target = _pick(pos)
    _dragging = false
    _drag_payload = {}
    if _press_target != null and _press_target.has_method("on_press_begin"):
        _drag_payload = _press_target.on_press_begin()

func _end_press(pos: Vector2) -> void:
    if _press_target == null:
        return
    var held := (Time.get_ticks_msec() / 1000.0) - _press_time
    if _dragging:
        var drop_target := _pick(pos)
        _press_target.on_drag_end(drop_target, _drag_payload)
    elif held >= HOLD_THRESHOLD:
        _press_target.on_hold(_drag_payload)
    else:
        _press_target.on_tap(_drag_payload)
    _press_target = null
    _dragging = false

func _pick(pos: Vector2) -> Node:
    # 复用 world.gd 的 _find_click_target 逻辑，改为返回 Node 而非直接 interact
    var world := get_tree().get_first_node_in_group("world")
    if world and world.has_method("find_click_target_node"):
        return world.find_click_target_node(pos)
    return null
```

**接入路径**：`world.gd` 新增 `find_click_target_node(pos) -> Node`（把现有 `_find_click_target` 的返回类型从 `WorldInteractable` 放宽到 `Node`，并**不调用 interact**），`_unhandled_input` 里删除左键直接 interact 的分支，改由 router 接管。

## 3. 手势语义表（唯一事实源）

| 手势 | 触发条件 | 语义 | 适用实体 |
|---|---|---|---|
| TAP | 左键按下→抬起，位移 < 8px，时长 < 0.35s | 主操作 | 全部 |
| SECONDARY | 右键按下 | 次操作（减量/取消） | 点餐台、托盘 |
| HOLD | 左键按住 ≥ 0.35s | 详情气泡 | 工位、点餐台 |

## B. 遗漏（两份方案都没覆盖）

### B1. 无"取消/回退"路径

### B2. 无"并发点击"防护
两份方案都没说：玩家在 100ms 内连点工位 3 次会怎样？主方案 `_flash_click` 会叠 3 个 tween；独立方案 `on_tap` 会开 3 个配方环。**返工要求**：`InteractionEntity` / `WorldInteractable` 子类必须有 `_action_lock: bool`，`on_tap` 入口 `if _action_lock: return`，动作结束（信号回调）才解锁。验收断言：连点 5 次，`KitchenManager.start_recipe` 只被调用 1 次。

### B4. 无"队列位与顾客实体解耦"的存档格式

### 3.2 `scripts/gameplay/click_gesture_router.gd`（新建，唯一新增节点）

```gdscript
class_name ClickGestureRouter
extends Node

signal gesture_dispatched(target: Node, gesture: int, payload: Dictionary)

enum Gesture { TAP, SECONDARY, HOLD, DRAG_DROP, CANCEL }

var _hold_threshold := 0.35
var _drag_threshold := 8.0
var _press_pos := Vector2.ZERO
var _press_time := 0.0
var _press_target: Node = null
var _dragging := false
var _drag_payload: Dictionary = {}

func _ready() -> void:
    process_mode = Node.PROCESS_MODE_PAUSABLE   # P1-B9：暂停时不响应
    _hold_threshold = PresentationManager.get_token_float("gesture.hold_threshold")
    _drag_threshold = PresentationManager.get_token_float("gesture.drag_threshold")

func _unhandled_input(event: InputEvent) -> void:
    if get_tree().paused:
        return
    if event is InputEventKey and event.pressed and event.keycode == KEY_ESCAPE:
        _dispatch_cancel()
        return
    if event is InputEventMouseButton:
        var mb := event as InputEventMouseButton
        if mb.button_index == MOUSE_BUTTON_LEFT:
            if mb.pressed: _begin_press(mb.position)
            else: _end_press(mb.position)
        elif mb.button_index == MOUSE_BUTTON_RIGHT and mb.pressed:
            _dispatch_secondary(mb.position)
    elif event is InputEventMouseMotion and _press_target != null:
        var mm := event as InputEventMouseMotion
        if not _dragging and mm.position.distance_to(_press_pos) > _drag_threshold:
            _dragging = true
            if _press_target.has_method("on_drag_begin"):
                _press_target.on_drag_begin(_drag_payload)

func _begin_press(pos: Vector2) -> void:
    _press_pos = pos
    _press_time = Time.get_ticks_msec() / 1000.0
    _press_target = _pick(pos)
    _dragging = false
    _drag_payload = {}
    if _press_target != null and _press_target.has_method("on_press_begin"):
        _drag_payload = _press_target.on_press_begin()

func _end_press(pos: Vector2) -> void:
    if _press_target == null:
        return
    var held := (Time.get_ticks_msec() / 1000.0) - _press_time
    if _dragging:
        var drop_target := _pick(pos)
        if _press_target.has_method("on_drag_end"):
            _press_target.on_drag_end(drop_target, _drag_payload)
    elif held >= _hold_threshold:
        if _press_target.has_method("on_hold"):
            _press_target.on_hold(_drag_payload)
    else:
        if _press_target.has_method("on_tap"):
            _press_target.on_tap(_drag_payload)
    _press_target = null
    _dragging = false

func _dispatch_secondary(pos: Vector2) -> void:
    var target := _pick(pos)
    if target == null:
        return
    # P1-D5：重新取 payload，不复用 _drag_payload
    var payload: Dictionary = {}
    if target.has_method("on_press_begin"):
        payload = target.on_press_begin()
    if target.has_method("on_secondary"):
        target.on_secondary(payload)

func _dispatch_cancel() -> void:
    # P1-B1：取消路径
    var world := get_tree().get_first_node_in_group("world")
    if world == null:
        return
    for child in world.get_children():
        if child.has_method("on_cancel"):
            child.on_cancel()

func _pick(pos: Vector2) -> Node:
    var world := get_tree().get_first_node_in_group("world")
    if world == null:
        push_error("[ClickGestureRouter] world 节点不存在")
        return null
    if not world.has_method("find_click_target_node"):
        push_error("[ClickGestureRouter] world.find_click_target_node 未实现，router 无法工作")
        return null
    return world.find_click_target_node(pos)
```
