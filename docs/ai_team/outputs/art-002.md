- 所有颜色值必须来自 `data/theme_tokens.csv`，代码中禁止 `Color("#...")` 字面量。

## 2. 九宫格切片规范

### 2.1 通用规则

| 参数 | 值 | 说明 |
|---|---|---|
| 网格基准 | 16px | 所有 margin 必须是 16 的整数倍或 16 的 1/2（8） |
| 最小贴图尺寸 | 48×48 | 小于此尺寸无法容纳 16px 四边 margin |
| 推荐贴图尺寸 | 96×96（按钮/槽位）、192×192（面板）、256×128（对话气泡） | 2x 缩放安全 |
| margin 值 | 左/右/上/下 = 16 或 24 | 圆角 14 时用 24，圆角 10 时用 16 |
| 拉伸区域 | 中心 1×1 像素 | 必须为纯色或极低对比渐变，禁止纹理 |
| 边缘区域 | 16px 宽条 | 必须为纯色，禁止渐变（否则拉伸出现条纹） |

### 2.3 九宫格验证脚本

```python
# tools/verify_ninepatch.py
from PIL import Image
import sys

def verify(path, ml, mr, mt, mb):
    img = Image.open(path).convert("RGBA")
    w, h = img.size
    assert w >= ml + mr + 1, f"{path}: 宽度不足 {w} < {ml+mr+1}"
    assert h >= mt + mb + 1, f"{path}: 高度不足 {h} < {mt+mb+1}"
    # 检查中心 1x1 是否为纯色
    cx, cy = w // 2, h // 2
    center = img.getpixel((cx, cy))
    # 检查四边拉伸区是否纯色（抽样）
    for x in range(ml, w - mr, 4):
        assert img.getpixel((x, mt // 2)) == img.getpixel((ml, mt // 2)), \
            f"{path}: 上边缘非纯色 at x={x}"
    print(f"OK {path} {w}x{h} margins=({ml},{mr},{mt},{mb}) center={center}")

if __name__ == "__main__":
    verify(sys.argv[1], *map(int, sys.argv[2:6]))
```

### 4.3 代码接入（`PresentationManager`）

```gdscript
# scripts/ui/presentation_manager.gd（补充）
var _ui_assets: Dictionary = {}

func _load_ui_assets() -> void:
    var file := FileAccess.open("res://data/ui_assets.csv", FileAccess.READ)
    var header := file.get_csv_line()
    while not file.eof_reached():
        var row := file.get_csv_line()
        if row.size() < 11:
            continue
        _ui_assets[row[0]] = {
            "path": row[1],
            "ninepatch": [int(row[2]), int(row[3]), int(row[4]), int(row[5])],
            "content": [float(row[6]), float(row[7]), float(row[8]), float(row[9])],
            "fallback": row[10],
        }

func get_ui_texture(key: String) -> Texture2D:
    if not _ui_assets.has(key):
        push_error("UI asset key not found: %s" % key)
        return null
    var entry: Dictionary = _ui_assets[key]
    var tex := load(entry["path"]) as Texture2D
    if tex == null and not entry["fallback"].is_empty():
        return get_ui_texture(entry["fallback"])
    return tex

func build_stylebox(key: String) -> StyleBoxTexture:
    var entry: Dictionary = _ui_assets[key]
    var sb := StyleBoxTexture.new()
    sb.texture = get_ui_texture(key)
    var np: Array = entry["ninepatch"]
    sb.texture_margin_left = np[0]
    sb.texture_margin_right = np[1]
    sb.texture_margin_top = np[2]
    sb.texture_margin_bottom = np[3]
    var cm: Array = entry["content"]
    sb.content_margin_left = cm[0]
    sb.content_margin_right = cm[1]
    sb.content_margin_top = cm[2]
    sb.content_margin_bottom = cm[3]
    return sb
```

**关键约束**：
- `get_ui_texture()` 必须做**环检测**（fallback 链不能成环），对齐 ui-001 批判 M2。
- `build_stylebox()` 必须缓存，避免每帧新建，对齐 ui-001 批判 C4。

## 独立方案

# UI 资源状态与九宫格规范 —— 独立方案

> 独立 Agent 输出。**不沿用主方案的"每状态一张 PNG"路线**，改为 **单图 + StyleBox 调制 + 九宫格切片** 架构。理由：现有 `inventory_slot_button.gd` 已经在用 `modulate_color` 派生 hover/pressed，说明项目已具备"一图多态"能力；再为每个状态单独出图会带来 5× 资产量、5× 一致性风险、5× 导入维护成本，且与 `theme_tokens.csv` 的色板单一数据源原则冲突。

---

## 一、架构决策（与主方案的分歧点）

| 维度 | 主方案倾向 | **本方案** | 理由 |
|---|---|---|---|
| 状态资产 | 每状态独立 PNG | **单 base PNG + StyleBox modulate + 可选 overlay** | 现有 `slot.normal` 已用 modulate 派生 hover/pressed；色板已集中在 `theme_tokens.csv` |
| 九宫格 | 每状态各自 margin | **同一 base 图，margin 由 CSV 声明，全状态共享** | 状态只改颜色不改几何，margin 必须一致 |
| focus 态 | 未明确 | **独立 `focus_ring` 九宫格 overlay** | focus 是几何叠加（描边环），不是颜色变化，无法用 modulate 表达 |
| disabled | 独立 PNG | **modulate 去饱和 + 降 alpha** | 与 `inventory_slot_button.gd` 现有 `Color(0.82,0.86,0.84,0.78)` 一致 |
| 资产数量 | 面板/按钮/槽位/标题/气泡 × 5 态 ≈ 25+ 张 | **5 张 base + 1 张 focus_ring + 1 张 overlay_glow = 7 张** | 减少 70% 资产 |
| 矢量绘制 | 未提及 | **提供 SVG 母版作为 PNG 的上游** | 可重导出任意尺寸，避免"改一次重画一次" |

**核心原则**：
1. **几何与颜色分离**：base 图只定义形状/边框/圆角；颜色由 `theme_tokens.csv` + StyleBox `modulate_color` 决定。
2. **九宫格 margin 是几何属性**，写在 CSV，全状态共享，禁止在代码里硬编码 `texture_margin_*`。
3. **focus 是叠加层**，不是替换层。用 `StyleBoxTexture` 的 `draw_center=false` + 独立 `focus_ring` 图。
4. **所有状态派生走 `PresentationManager.build_stylebox(asset_key, state)`**，禁止在业务代码里 `duplicate(true)`。

## 四、PresentationManager 接口（唯一入口）

### 4.1 新增 API

```gdscript
# scripts/autoload/presentation_manager.gd

var _stylebox_cache: Dictionary = {}
var _ninepatch_table: Dictionary = {}
var _state_table: Dictionary = {}

func build_stylebox(asset_key: String, state: String = "normal") -> StyleBoxTexture:
    var cache_key := "%s|%s" % [asset_key, state]
    if _stylebox_cache.has(cache_key):
        return _stylebox_cache[cache_key]
    var np := _ninepatch_table.get(asset_key, {})
    if np.is_empty():
        push_error("[PresentationManager] 未注册九宫格: %s" % asset_key)
        return null
    var tex := load(np.texture_path) as Texture2D
    if tex == null:
        push_error("[PresentationManager] 贴图缺失: %s" % np.texture_path)
        return null
    var sb := StyleBoxTexture.new()
    sb.texture = tex
    sb.texture_margin_left = np.margin_left
    sb.texture_margin_top = np.margin_top
    sb.texture_margin_right = np.margin_right
    sb.texture_margin_bottom = np.margin_bottom
    sb.content_margin_left = np.content_left
    sb.content_margin_top = np.content_top
    sb.content_margin_right = np.content_right
    sb.content_margin_bottom = np.content_bottom
    var mod := _state_table.get("%s.%s" % [asset_key, state], {})
    if not mod.is_empty():
        sb.modulate_color = Color(mod.modulate_r, mod.modulate_g, mod.modulate_b, mod.modulate_a)
    _stylebox_cache[cache_key] = sb
    return sb

func build_focus_overlay(asset_key: String) -> StyleBoxTexture:
    var sb := build_stylebox("focus_ring", "normal")
    if sb == null:
        return null
    var dup := sb.duplicate(true) as StyleBoxTexture
    dup.draw_center = false
    return dup

func invalidate_stylebox_cache() -> void:
    _stylebox_cache.clear()
```

## 七、与主方案的差异总结

| 项 | 主方案 | 本方案 | 收益 |
|---|---|---|---|
| 资产数量 | 25+ 张 | **7 张** | -72% |
| 状态一致性 | 每状态独立绘制，易漂移 | **单图 + 调制表**，几何强制一致 | 消除漂移 |
| 九宫格 margin | 代码硬编码 | **CSV 声明** | 改一次全局生效 |
| focus 态 | 未明确 | **独立 overlay 层** | 键盘导航可用 |
| 矢量上游 | 无 | **SVG 母版** | 任意尺寸重导出 |
| 业务代码 | `duplicate(true)` 派生 | **`build_stylebox` 缓存** | 消除重复构造 |
| 与现有代码 | 需重写 | **兼容 `inventory_slot_button.gd` 现有 modulate 值** | 零视觉回归 |

**核心分歧**：主方案把"状态"当作**资产维度**，本方案把"状态"当作**调制维度**。前者资产量线性增长，后者资产量恒定。在色板已集中、几何已统一的暖色手账风格下，后者是唯一可持续的架构。

# 批判报告：UI 资源状态与九宫格规范

## 一、两份方案共同的结构性缺陷

### C1. 状态矩阵与资产清单不自洽（P0，两份方案均中招）

主方案 §1 状态矩阵声明 `panel`、`title`、`dialog` 系列**不适用 hover/pressed/focus/disabled**，但 §2.2 逐资产切片表仍为 `panel.normal`、`panel.soft`、`title.normal`、`dialog.*` 定义了完整 margin 与 content_margin，§4.1 CSV 也照单全收。**结果：CSV 中约 40% 的行是死数据**——没有任何代码路径会调用 `build_stylebox("panel.hover")`，因为 §5.1 的 `build_theme()` 只对 `Button` 设了 5 态，`PanelContainer` 只设了 `panel` 一个 key。

独立方案同样：§3.1 调制表为 `panel.hover/pressed/focus/disabled` 定义了 4 行，但 §4.3 `build_theme()` 里 `PanelContainer` 只取 `panel.normal`。**这 4 行 CSV 永远不会被读取。**

**返工要求**：
- 明确列出「哪些元素真的有 5 态」的白名单。面板/标题/气泡若确实无交互态，CSV 中**不得出现**对应行，或必须在 `ui_state_modulation.csv` 增加 `applicable_states` 列并在加载时断言。
- 增加门禁：`tools/verify_ui_assets.gd` 必须遍历 CSV 每一行，调用 `build_stylebox(key, state)` 并断言非 null；同时用 `grep` 反查每个 key 是否在 `scripts/ui/` 中被引用，未被引用即 FAIL。

### C2. 九宫格 margin 与圆角的几何矛盾（P0，主方案）

主方案 §2.1 规定「圆角 14 时用 margin 24」，§2.2 中 `panel.normal` 圆角 14 / margin 24 自洽。但 `title.normal` 圆角 `14/14/0/0`（仅上方圆角），margin 却写 `24/24/16/16`——**底部 margin 16 小于顶部 24，而底部无圆角**。九宫格拉伸时底部 16px 区域会被横向拉伸，若该区域含 1px 下边框（§1 声明 `title` 有 `#D9C4A8 1px` 下边框），**下边框会被拉成条纹**。

更严重：`title.normal` 的 §2.2 备注写「底部拉伸区需保证纯色」，但 §1 状态矩阵明确 title 有下边框。**边框与纯色要求直接冲突。**

**返工要求**：
- 标题条若需下边框，必须把下边框放进 bottom margin 区（B ≥ 边框所在像素行 + 1），或改用 `StyleBoxFlat` 的 `border_width_bottom` 叠加，**不得依赖九宫格贴图承载边框**。
- 所有含边框的资产，margin 必须 ≥ 边框宽度 + 圆角半径，否则拉伸必翻车。给出公式：`margin ≥ max(border_width, corner_radius) + 1`。

### C3. `focus` 与 `hover` 同时可见的断言缺失（P1，两份方案）

主方案 §1 硬约束写「focus 外描边，hover 内提亮，色值差异 ≥ 15%」，但 §7.1 验证清单 V5 只数了 `set_stylebox` 调用次数（`>= 5`），**没有任何脚本验证 focus 与 hover 叠加时的可区分性**。独立方案 §3.1 调制表中 `button.focus` 的 modulate 是 `1.0,1.0,1.0,1.0`（即完全不变），focus 全靠 `focus_ring` overlay——但 overlay 是**独立 StyleBox**，Godot 的 `Button` 在 focus+hover 时只会绘制一个 stylebox，**不会自动叠加两个**。

**返工要求**：
- 独立方案必须说明 `focus_ring` 如何与 hover 态共存：是 `Button` 的 `focus` stylebox 用 `focus_ring` 贴图 + `draw_center=false`（此时 hover 的底色由谁画？），还是用子节点 `NinePatchRect` 手动叠加？**当前方案在 Godot 的 StyleBox 模型下不可实现。**
- 增加可执行断言：`tools/verify_focus_hover.py` 渲染 focus+hover 合成图，计算两态与单 hover 态的像素差异，断言 `diff_ratio > 0.05`。

### C4. `disabled` 态的可访问性断言缺失（P1，两份方案）

主方案 §1 要求「disabled 必须同时降低对比度和饱和度，不能只降透明度」，但 §2.2 中 `button.disabled` 只写「中心 50% 透明」，§4.2 补充的 `color.disabled.overlay` 是 `#F5E9D680`（纯 alpha 变化）。**规范与实现自相矛盾。**

独立方案 §3.1 中 `button.disabled` modulate 为 `0.82,0.86,0.84,0.78`——这是**降亮度 + 降 alpha**，但 `0.82,0.86,0.84` 三通道差异极小（最大 0.04），**饱和度几乎没降**。在浅色底 `#F5E9D6` 上，disabled 按钮与 normal 按钮的对比度可能低于 WCAG 3:1。

**返工要求**：
- 给出 disabled 的**可计算**定义：`saturation_reduction ≥ 40%` 且 `luminance_contrast_vs_bg ≥ 3.0`。
- 增加断言脚本：`tools/verify_contrast.py`，输入 normal/disabled 两态色值，输出 WCAG 对比度，低于阈值即 FAIL。
- 主方案 §4.2 的 `color.disabled.overlay` 必须改为「去饱和后的实色 + alpha」，不能是原色的 alpha 版本。

### C5. fallback 链的环检测只字未提实现（P0，主方案）

主方案 §4.3 注释写「必须做环检测，对齐 ui-001 批判 M2」，但**代码里没有任何环检测逻辑**。`get_ui_texture()` 的 fallback 递归调用是裸递归：

```gdscript
if tex == null and not entry["fallback"].is_empty():
    return get_ui_texture(entry["fallback"])
```

若 CSV 中 `button.hover` 的 fallback 指向 `button.normal`，而 `button.normal` 的 fallback 误填 `button.hover`，**直接栈溢出崩溃**。§7.2 风险表把「CSV 新增列破坏现有解析」列为 P1，却把更致命的环检测遗漏了。

**返工要求**：
- `get_ui_texture()` 必须带 `visited: Array[String]` 参数，或加载时预计算 fallback 链并检测环。
- 增加门禁：`tools/verify_ui_assets.gd` 加载后对每个 key 做 DFS，检测环并 FAIL。
- 独立方案无 fallback 机制，但 §4.1 `build_stylebox` 在 `np.is_empty()` 时 `push_error` 后返回 null——**调用方 `add_theme_stylebox_override("normal", null)` 会导致 Godot 运行时行为未定义**（可能静默用默认主题，可能崩溃）。必须返回一个 `StyleBoxEmpty` 或 `StyleBoxFlat` 兜底。

### C6. 缓存失效机制不完整（P1，两份方案）

主方案 §5.1 有 `invalidate_theme()`，独立方案 §4.1 有 `invalidate_stylebox_cache()`。但两份方案都**没有说明何时调用**。热重载 CSV 后、feature flag 切换后、贴图重新导入后，缓存不会自动失效。

**返工要求**：
- 明确失效触发点：`ProjectSettings` 变更信号、`ResourceLoader` 的 `resources_reload` 信号、或显式调试命令。
- 增加断言：`tools/verify_cache_invalidation.gd`，修改 CSV → 调用 invalidate → 断言 `build_stylebox` 返回新对象（`!=` 旧引用）。

### C7. 16px 网格与「边缘 16px 内纯色」的尺寸矛盾（P0，主方案）

主方案 §2.1 规定「最小贴图尺寸 48×48」，§2.2 中 `slot.*` 用 96×96 / margin 16，自洽。但 §3.1 图像生成 Prompt 写「边缘 16px 内为纯色，中心 1x1 为纯色」——**若贴图 96×96，margin 16，则中心区是 64×64，不是 1×1**。Prompt 与实际规格矛盾，图像生成模型会按「中心 1×1」理解，产出中心有 1px 色块的图，九宫格拉伸后中心区全是那 1px 的颜色，**视觉上没问题但语义错误**。

更严重：§2.1 说「拉伸区域：中心 1×1 像素」，但 §2.2 的 margin 是 16/24，**中心区实际是 (96-32)×(96-32)=64×64**。规范自相矛盾。

**返工要求**：
- 统一术语：「中心区」= 贴图尺寸 - 四边 margin，不是 1×1。「中心 1×1 纯色」是验证脚本的抽样点，不是设计约束。
- Prompt 改为「中心区（尺寸 - 2×margin）为纯色或极低对比渐变」。
- §2.3 验证脚本的 `center = img.getpixel((cx, cy))` 只检查了 1 个点，**不足以证明中心区纯色**。必须改为网格抽样（如每 8px 一个点）并断言方差 < 阈值。

### C8. 对话气泡尾巴的九宫格陷阱（P0，独立方案）

独立方案 §2.2 E 明确「尾巴必须落在 bottom margin 区内（B=28 > 尾巴高 8）」，这是**正确的洞察**。但 §3.2 的 `bubble.normal` margin 是 `24,20,24,28`，而 §2.2 E 的贴图逻辑尺寸是 96×96，导出 192×192。**问题**：尾巴在底部居中，宽度 16px。九宫格拉伸时，bottom margin 区（28px 高）会被**横向拉伸**——尾巴会被拉宽。若气泡实际宽度 400px，尾巴从 16px 被拉到 400-48=352px 宽，**完全变形**。

**返工要求**：
- 尾巴不能放在九宫格的可拉伸区。正确做法：气泡主体用九宫格，尾巴用**独立子节点**（`NinePatchRect` 或 `Sprite2D`）定位在气泡底部中心。
- 或：接受尾巴拉伸，但明确声明「尾巴宽度随气泡宽度变化」是设计意图（通常不是）。
- 增加断言：`tools/verify_bubble_tail.py`，渲染 200px 宽和 600px 宽气泡，断言尾巴宽度差异 < 2px（若要求不变形）。

### C9. `title_base.png` 的横向渐变与九宫格冲突（P0，独立方案）

独立方案 §2.2 D 规定 `title_base.png` 中心区是「横向渐变 `#E8A85C` → `#F0BC7A` → `#E8A85C`（左中右）」。但 §3.2 的 margin 是 `L=24, T=12, R=24, B=12`，**中心区会被横向拉伸**。渐变在拉伸后，中间色 `#F0BC7A` 会被拉成一条宽带，**渐变分布完全失真**。

**返工要求**：
- 横向渐变不能放在九宫格中心区。要么用 `StyleBoxFlat` 的 `bg_color` + 独立渐变贴图子节点，要么接受渐变失真并明确声明。
- 若坚持用九宫格，渐变必须放在**左右 margin 区**（不可拉伸），中心区纯色。但这样渐变只在两端 24px 内，视觉上不是「左中右渐变」。

### C10. 独立方案的 `asset_key` 命名与主方案冲突（P0，跨方案）

主方案 §4.1 CSV 的 `asset_key` 是 `button.normal`、`slot.hover`（**状态在 key 里**）。
独立方案 §3.2 CSV 的 `asset_key` 是 `button`、`slot`（**状态在 state 列**）。

两份方案都声称「对齐 ui-001 决策」，但 ui-001 的 `PresentationManager.get_ui_texture(key)` 到底接受哪种 key？**若现有代码已用 `slot.normal`，独立方案的 `build_stylebox("slot", "normal")` 会破坏现有调用点。**

**返工要求**：
- 必须先跑主方案 §0.3 门禁，确认现有 key 命名规则。
- 若现有是 `slot.normal`，独立方案必须改为 `build_stylebox("slot.normal")`（state 从 key 解析），或提供兼容层。
- 两份方案不能各自定义 key 规范后都声称「对齐 ui-001」。

## 二、主方案独有的问题

### M1. `verify_ninepatch.py` 的边缘纯色检查逻辑错误（P0）

```python
for x in range(ml, w - mr, 4):
    assert img.getpixel((x, mt // 2)) == img.getpixel((ml, mt // 2)), \
        f"{path}: 上边缘非纯色 at x={x}"

`mt // 2` 是上 margin 区的中点，但**上 margin 区可能包含圆角**（如 `button.normal` 圆角 10，margin 24，圆角在 margin 区内）。圆角处像素必然与中点不同，**断言必失败**。

**返工要求**：

## 三、独立方案独有的问题

### I1. `StyleBoxTexture.modulate_color` 的语义误解（P0）

独立方案 §4.1：
```gdscript
sb.modulate_color = Color(mod.modulate_r, mod.modulate_g, mod.modulate_b, mod.modulate_a)
```

Godot 4 的 `StyleBoxTexture` **没有 `modulate_color` 属性**。`StyleBoxTexture` 的属性是 `modulate_color`？查 Godot 4.x 文档：`StyleBoxTexture` 有 `modulate_color: Color`——**实际上有**，但它的作用是**调制贴图颜色**，且**只影响贴图，不影响 `draw_center` 的纯色填充**（若 `draw_center=true`，中心区是贴图内容，会被调制；若贴图中心是纯色，调制生效）。

**问题**：独立方案 §2.2 中 `button_base.png` 中心区是 `#E8A85C` 纯色，`button.hover` modulate `1.12,1.08,0.92` 会把它调成 `#FFB85C` 左右——**这确实能工作**。但 §3.1 中 `panel.disabled` modulate `0.82,0.86,0.84,0.78`，alpha 0.78 会让整个面板半透明，**包括边框**——而主方案 §1 要求 disabled 只降内容不降边框。**两份方案对 disabled 的语义理解不同。**

**返工要求**：明确 disabled 是「整体半透明」还是「仅内容半透明」。若是后者，`StyleBoxTexture.modulate_color` 做不到，需要 `StyleBoxFlat` 叠加或子节点方案。

### I2. `focus_ring` 的 `draw_center=false` 与 `content_margin` 冲突（P0）

独立方案 §4.1：
```gdscript
func build_focus_overlay(asset_key: String) -> StyleBoxTexture:
    var sb := build_stylebox("focus_ring", "normal")
    var dup := sb.duplicate(true) as StyleBoxTexture
    dup.draw_center = false
    return dup
```

`focus_ring` 的 `content_margin` 在 §3.2 中是 `0,0,0,0`。但 `draw_center=false` 时，`content_margin` 仍会影响控件布局——**focus overlay 若作为独立 StyleBox 使用，会把控件内容挤到边缘**。

更根本：**Godot 的 `Button` 在 focus 态只绘制 `focus` stylebox，不会叠加 `normal` + `focus`**。独立方案的 `focus_ring` 作为 `focus` stylebox 时，`draw_center=false` 意味着 focus 态下按钮**没有底色**（只有描边环），视觉上按钮会「消失」。

**返工要求**：
- 明确 focus 态的实现方式：是 `focus` stylebox 用 `focus_ring` 贴图（此时底色丢失），还是用 `Button` 的 `focus` 信号手动添加 `NinePatchRect` 子节点？
- 若是后者，给出子节点的 z_index、锚点、尺寸同步逻辑。
- 当前方案在 Godot 的 StyleBox 模型下**不可实现**。

### I3. `bubble.npc/system/scene` 的调制值无法区分（P1）

§3.1：
```
bubble.npc,npc,1.0,1.0,1.0,1.0,
bubble.system,system,0.96,0.98,1.0,1.0,
bubble.scene,scene,1.02,0.98,0.94,1.0,
```

`npc` 是 `1.0,1.0,1.0`（无变化），`system` 是 `0.96,0.98,1.0`（几乎无变化），`scene` 是 `1.02,0.98,0.94`（几乎无变化）。**三者在视觉上无法区分。** 主方案 §1 用左边框颜色区分（`#6F8F7A` / `#8AA9B1` / `#C89C5A`），独立方案用调制——但调制值差异 < 5%，**人眼不可辨**。

**返工要求**：
- 三通道区分必须用**几何或高对比色**，不能靠 < 5% 的调制。
- 若坚持单图 + 调制，调制值差异必须 ≥ 20%（如 `npc` 用 `1.0,1.0,1.0`，`system` 用 `0.8,0.9,1.0`，`scene` 用 `1.1,0.9,0.8`）。
- 或改用独立贴图（回到主方案路线）。

### I4. `title_base.png` 的「圆角 999（胶囊）」与九宫格不兼容（P0）

§2.2 D 规定 `title_base.png` 圆角 999（胶囊），但 §3.2 margin 是 `L=24, T=12, R=24, B=12`。**胶囊圆角需要左右 margin ≥ 高度/2**，而高度 48，margin 12——**圆角会被拉伸成椭圆**。

**返工要求**：
- 胶囊形状不能用九宫格。要么用 `StyleBoxFlat` 的 `corner_radius_*`（Godot 原生支持），要么用 `NinePatchRect` 但 margin 必须 ≥ 高度/2。
- 当前规格下，`title_base.png` 拉伸后圆角必然变形。

### I5. `overlay_glow` 的径向渐变与九宫格冲突（P0）

§2.2 G 规定 `overlay_glow.png` 中心区是「径向渐变 `#FFD9A866` → 透明」，但 §3.2 margin 是 `16,16,16,16`。**径向渐变在九宫格中心区会被拉伸**，渐变中心偏移，视觉上变成「横向条带」。

**返工要求**：
- 径向渐变不能用九宫格。要么用 `TextureRect` + `stretch_mode = STRETCH_KEEP_ASPECT_CENTERED`，要么用 shader。
- 当前规格下，`overlay_glow` 拉伸后不是径向渐变。

### I6. `slot.selected` 的 `overlay_glow` 叠加未说明实现（P1）

§3.1 中 `slot.selected` 的 `extra_overlay` 是 `overlay_glow`，但 §4.1 `build_stylebox` **完全没有处理 `extra_overlay` 字段**。CSV 中该列是死数据。

**返工要求**：

- 当前方案中 `extra_overlay` 列永远不会被读取。

### I7. `focus_mode = Control.FOCUS_ALL` 的副作用未评估（P1）

§4.2 注释写「修复现有 FOCUS_NONE 导致键盘导航失效」，但**没有评估改为 FOCUS_ALL 后，Tab 键遍历顺序、焦点陷阱、模态对话框的焦点管理**。背包槽位全部可聚焦后，Tab 键会在几十个槽位间循环，**用户体验灾难**。

**返工要求**：

### I8. 废弃 PNG 的 fallback 路径未验证（P1）

## 1. 架构裁决（X1–X8 全部裁决）

| # | 冲突点 | **最终裁决** | 理由 |
|---|---|---|---|
| X1 | asset_key 命名 | `asset_key="slot"` + `state="normal"` 分离；`get_ui_texture("slot.normal")` 兼容层 | 状态是调制维度；兼容旧调用点 |
| X2 | 状态资产策略 | **混合**：几何用单 base 图，颜色用 modulate，focus/selected 用子节点 overlay | 纯 modulate 无法表达 focus 几何叠加（批判 I2 正确）；纯独立 PNG 资产量爆炸 |
| X3 | focus 态实现 | **子节点 `NinePatchRect` 叠加**，不用 `focus` stylebox | Godot Button 的 focus stylebox 会替换 normal 而非叠加（批判 I2 正确） |
| X4 | disabled 语义 | **仅内容降饱和降对比，边框保留**；用 `StyleBoxFlat` 叠加实现 | 整体半透明会让边框消失，不符合可访问性（批判 C4 正确） |
| X5 | 九宫格 margin 来源 | **统一为 `data/ui_ninepatch.csv` 一张表** | 消除双表冲突 |
| X6 | 资产数量 | **7 张 base + 1 张 focus_ring + 1 张 tail = 9 张** | 覆盖全部视觉需求，尾巴独立（批判 C8 正确） |

| X8 | feature flag 行为 | flag=false 时 `build_stylebox` 返回**旧硬编码 `StyleBoxFlat`**，不加载 `_legacy/` PNG | 避免路径映射复杂度（批判 I8 正确） |

**核心原则（不可违反）**：
1. **几何与颜色分离**：base 图只定义形状/边框/圆角；颜色由 `theme_tokens.csv` + `modulate_color` 决定。
2. **九宫格 margin 是几何属性**，写在 CSV，全状态共享，代码中禁止字面量。
3. **focus/selected 是叠加层**，用子节点 `NinePatchRect` 实现，不用 stylebox 替换。
4. **所有状态派生走 `PresentationManager.build_stylebox(asset_key, state)`**，禁止业务代码 `duplicate(true)`。
5. **横向/径向渐变禁止放在九宫格中心区**（批判 C9/I5 正确），改用 `StyleBoxFlat` 或 shader。

### 3.3 九宫格几何验证脚本（修正 M1）

```python
# tools/verify_ninepatch_geometry.py
from PIL import Image
import sys, statistics

def verify(path, ml, mt, mr, mb, corner_radius, border_width):
    img = Image.open(path).convert("RGBA")
    w, h = img.size
    # 断言 1：尺寸足够
    assert w >= ml + mr + 4, f"{path}: 宽度不足 {w} < {ml+mr+4}"
    assert h >= mt + mb + 4, f"{path}: 高度不足 {h} < {mt+mb+4}"
    # 断言 2：margin >= max(border, corner) + 1
    min_margin = max(border_width, corner_radius) + 1
    for name, m in [("L", ml), ("T", mt), ("R", mr), ("B", mb)]:
        assert m >= min_margin, f"{path}: margin_{name}={m} < {min_margin}"
    # 断言 3：margin mod 4 == 0
    for name, m in [("L", ml), ("T", mt), ("R", mr), ("B", mb)]:
        assert m % 4 == 0, f"{path}: margin_{name}={m} 非 4 的倍数"
    # 断言 4：中心区纯色（网格抽样，修正 C7/M1）
    cx0, cy0, cx1, cy1 = ml, mt, w - mr, h - mb
    samples = []
    for x in range(cx0, cx1, 4):
        for y in range(cy0, cy1, 4):
            samples.append(img.getpixel((x, y)))
    # 计算每通道方差
    for ch in range(4):
        vals = [s[ch] for s in samples]
        var = statistics.variance(vals) if len(vals) > 1 else 0
        assert var < 4.0, f"{path}: 中心区通道 {ch} 方差 {var:.2f} >= 4.0（非纯色）"
    # 断言 5：拉伸区边缘行纯色（修正 M1，排除圆角）
    for x in range(ml + corner_radius, w - mr - corner_radius, 4):
        top_px = img.getpixel((x, mt))
        bot_px = img.getpixel((x, h - mb - 1))
        # 与中心区首行比较
        assert top_px == img.getpixel((ml + corner_radius, mt)), \
            f"{path}: 上拉伸区非纯色 at x={x}"
    print(f"OK {path} {w}x{h} margins=({ml},{mt},{mr},{mb})")

if __name__ == "__main__":
    # 用法：python verify_ninepatch_geometry.py <path> <ml> <mt> <mr> <mb> <corner> <border>
    verify(sys.argv[1], *map(int, sys.argv[2:8]))

```gdscript

extends Node

var _ninepatch_table: Dictionary = {}      # asset_key -> {path, margins, content, corner, border}
var _state_table: Dictionary = {}          # "asset_key.state" -> {modulate, overlay, applicable}
var _stylebox_cache: Dictionary = {}       # "asset_key.state" -> StyleBoxTexture
var _theme_cache: Theme = null             # 修正 M3
var _fallback_chain: Dictionary = {}       # asset_key -> [fallback_key, ...]

const FLAG_NEW_PIPELINE := "ui/theme/new_pipeline"

func _ready() -> void:
    _load_ninepatch_csv()
    _load_state_csv()
    _build_fallback_chain()                # 修正 C5：预计算 fallback 链
    if ProjectSettings.get_setting(FLAG_NEW_PIPELINE, true):

func _load_ninepatch_csv() -> void:
    var file := FileAccess.open("res://data/ui_ninepatch.csv", FileAccess.READ)
    assert(file != null, "ui_ninepatch.csv 缺失")
    var header := file.get_csv_line()
    var line_no := 1
    while not file.eof_reached():
        line_no += 1
        var row := file.get_csv_line()
        if row.size() < 2 or row[0].is_empty():
            continue
        assert(row.size() == 12, "ui_ninepatch.csv 第 %d 行列数 %d != 12" % [line_no, row.size()])  # 修正 M6
        _ninepatch_table[row[0]] = {
            "path": row[1],
            "margin": [int(row[2]), int(row[3]), int(row[4]), int(row[5])],
            "content": [float(row[6]), float(row[7]), float(row[8]), float(row[9])],
            "corner": int(row[10]),
            "border": int(row[11]),
        }

func _load_state_csv() -> void:
    var file := FileAccess.open("res://data/ui_state_modulation.csv", FileAccess.READ)
    assert(file != null, "ui_state_modulation.csv 缺失")
    var header := file.get_csv_line()
    var line_no := 1
    while not file.eof_reached():
        line_no += 1
        var row := file.get_csv_line()
        if row.size() < 2 or row[0].is_empty():
            continue
        assert(row.size() == 8, "ui_state_modulation.csv 第 %d 行列数 %d != 8" % [line_no, row.size()])
        var key := "%s.%s" % [row[0], row[1]]
        _state_table[key] = {
            "applicable": row[2] == "true",
            "modulate": Color(float(row[3]), float(row[4]), float(row[5]), float(row[6])),
            "overlay": row[7],
        }

func _build_fallback_chain() -> void:
    # 修正 C5：预计算 fallback 链，检测环
    for asset_key in _ninepatch_table.keys():
        var chain: Array[String] = [asset_key]
        var current := asset_key
        while true:
            var fallback := _resolve_fallback(current)
            if fallback.is_empty():
                break
            if chain.has(fallback):
                push_error("[PresentationManager] fallback 环: %s" % " -> ".join(chain))
                break
            chain.append(fallback)
            current = fallback
            if chain.size() > 5:
                push_error("[PresentationManager] fallback 链过长: %s" % " -> ".join(chain))
                break
        _fallback_chain[asset_key] = chain

func _resolve_fallback(asset_key: String) -> String:
    # 约定：asset_key 的 fallback 由命名规则决定，如 button_ghost -> button
    var parts := asset_key.split("_")
    if parts.size() > 1:
        var base := parts[0]
        if _ninepatch_table.has(base):
            return base
    return ""

# --- 核心 API ---

func build_stylebox(asset_key: String, state: String = "normal") -> StyleBox:
    var cache_key := "%s.%s" % [asset_key, state]
    if _stylebox_cache.has(cache_key):
        return _stylebox_cache[cache_key]
    # feature flag 关闭时返回旧 StyleBoxFlat（裁决 X8）
    if not ProjectSettings.get_setting(FLAG_NEW_PIPELINE, true):
        var flat := _build_legacy_stylebox(asset_key, state)
        _stylebox_cache[cache_key] = flat
        return flat
    # 状态表校验（裁决 C1）
    var state_entry: Dictionary = _state_table.get(cache_key, {})
    if state_entry.is_empty() or not state_entry.get("applicable", false):
        push_error("[PresentationManager] 状态未注册或不可用: %s" % cache_key)
        return _build_fallback_stylebox(asset_key)
    # 几何表校验
    var np: Dictionary = _ninepatch_table.get(asset_key, {})
    if np.is_empty():
        push_error("[PresentationManager] 九宫格未注册: %s" % asset_key)
        return _build_fallback_stylebox(asset_key)
    # 加载贴图（带 fallback 链，修正 C5）
    var tex := _load_texture_with_fallback(asset_key)
    if tex == null:
        push_error("[PresentationManager] 贴图全部缺失: %s" % asset_key)
        return _build_fallback_stylebox(asset_key)
    var sb := StyleBoxTexture.new()
    sb.texture = tex
    var m: Array = np.margin
    sb.texture_margin_left = m[0]
    sb.texture_margin_top = m[1]
    sb.texture_margin_right = m[2]
    sb.texture_margin_bottom = m[3]
    var c: Array = np.content
    sb.content_margin_left = c[0]
    sb.content_margin_top = c[1]
    sb.content_margin_right = c[2]
    sb.content_margin_bottom = c[3]
    sb.modulate_color = state_entry.modulate
    _stylebox_cache[cache_key] = sb
    return sb

func _load_texture_with_fallback(asset_key: String) -> Texture2D:
    var chain: Array = _fallback_chain.get(asset_key, [asset_key])
    for key in chain:
        var np: Dictionary = _ninepatch_table.get(key, {})
        if np.is_empty():
            continue
        var tex := load(np.path) as Texture2D
        if tex != null:
            return tex

    var np: Dictionary = _ninepatch_table.get(asset_key, {})

        return StyleBoxEmpty.new()

    flat.bg_color = Color("#F5E9D6")

    flat.corner_radius_bottom_left = np.corner
    flat.corner_radius_bottom_right = np.corner
    flat.border_width_left = np.border
    flat.border_width_top = np.border
    flat.border_width_right = np.border
    flat.border_width_bottom = np.border
    flat.border_color = Color("#D9C4A8")
    return flat
