# HUD 信息层级与主题接口重构方案

> 目标：左上角不再拥挤；提示永不重叠；按钮状态齐全；所有 UI 资源经 `PresentationManager` + CSV 稳定 key 接入；低饱和暖色、柔光、整洁但不冷清。

---

## 0. 问题诊断（先对齐现状）

| 现象 | 根因 | 涉及文件 |
|---|---|---|
| 左上角过杂 | `_status_panel` / `_calendar_panel` / `_context_panel` / `_npc_hint_panel` 全部堆在左上，各自独立定位 | `scripts/ui/hud.gd` |
| 提示重叠 | `_notice_panel`、`_npc_hint_panel`、`_pointer_panel` 各自 `visible` 切换，无统一调度 | `hud.gd` + `NoticeManager` |
| 按钮状态缺失 | `_make_menu_button` 只设 text/size，无 hover/pressed/disabled 视觉；HUD 内按钮同样 | `main_menu.gd`、`hud.gd` |
| 主题散落 | `main_menu.gd` 里 `Theme.new()` 手搓；`hud.gd` 无 Theme；颜色硬编码 `Color("#f1d08b")` | 两处 |
| 资源未走 PM | 字体、颜色、图标多处硬编码，未走 `PresentationManager.get_token/get_color` | 全局 |

## 5. PresentationManager API 扩展

**文件**：`autoload/presentation_manager.gd`

新增方法（保留现有全部接口，向后兼容）：

```gdscript
# --- 主题 token ---
func get_ui_color(token: String, fallback: Color = Color.WHITE) -> Color:
    return get_color(token, fallback)

func get_ui_radius(token: String, fallback: int = 8) -> int:
    return int(get_token(token, str(fallback)))

func get_ui_space(token: String, fallback: int = 8) -> int:
    return int(get_token(token, str(fallback)))

# --- 文本样式 ---
func get_text_style(style_id: String) -> Dictionary:
    return ConfigDB.get_row("ui_text_styles", style_id)

func apply_text_style(label: Label, style_id: String) -> void:
    var s := get_text_style(style_id)
    if s.is_empty(): return
    var font := SystemFont.new()
    font.font_names = get_font_names(str(s.get("font_role", "font.body")))
    label.add_theme_font_override("font", font)
    label.add_theme_font_size_override("font_size", int(s.get("font_size", 18)))
    label.add_theme_color_override("font_color", get_color(str(s.get("color_token", "color.text.primary"))))
    var oc := str(s.get("outline_color_token", ""))
    if oc != "":
        label.add_theme_color_override("font_outline_color", get_color(oc))
        label.add_theme_constant_override("outline_size", int(s.get("outline_size", 0)))
    label.add_theme_constant_override("line_spacing", int(s.get("line_spacing", 4)))

# --- UI 贴图 ---
func get_ui_texture(asset_id: String) -> Texture2D:
    var row := ConfigDB.get_row("ui_assets", asset_id)
    if row.is_empty(): return null
    var path := str(row.get("path", ""))
    var tex := _load_texture(path)
    if tex == null:
        var fb := str(row.get("fallback_asset_id", ""))
        if fb != "": return get_ui_texture(fb)
    return tex

# --- 统一 Theme 构建 ---
func build_ui_theme() -> Theme:
    var theme := Theme.new()
    var body_font := SystemFont.new()
    body_font.font_names = get_font_names("font.body")
    theme.default_font = body_font
    theme.default_font_size = 20

    # Button 四态
    for variant in ["primary", "ghost"]:
        var normal := _make_button_stylebox("button.%s.normal" % variant)
        var hover := _make_button_stylebox("button.%s.hover" % variant)
        var pressed := _make_button_stylebox("button.%s.pressed" % variant)
        var disabled := _make_button_stylebox("button.%s.disabled" % variant)
        theme.set_stylebox("normal", "Button", normal) if variant == "primary" else null
        # 用类型变体：Button/PrimaryButton、Button/GhostButton
        var type_name := "PrimaryButton" if variant == "primary" else "GhostButton"
        theme.set_type_variation(type_name, "Button")
        theme.set_stylebox("normal", type_name, normal)
        theme.set_stylebox("hover", type_name, hover)
        theme.set_stylebox("pressed", type_name, pressed)
        theme.set_stylebox("disabled", type_name, disabled)
        theme.set_stylebox("focus", type_name, _make_focus_stylebox())
        theme.set_color("font_color", type_name, get_color("color.text.primary"))
        theme.set_color("font_hover_color", type_name, get_color("color.text.primary"))
        theme.set_color("font_pressed_color", type_name, get_color("color.text.on_accent"))
        theme.set_color("font_disabled_color", type_name, get_color("color.text.muted"))

    # PanelContainer
    theme.set_stylebox("panel", "PanelContainer", _make_panel_stylebox("panel.paper"))
    theme.set_type_variation("SoftPanel", "PanelContainer")
    theme.set_stylebox("panel", "SoftPanel", _make_panel_stylebox("panel.paper_soft"))

    # Label
    theme.set_color("font_color", "Label", get_color("color.text.primary"))

    return theme

func _make_button_stylebox(asset_id: String) -> StyleBox:
    var tex := get_ui_texture(asset_id)
    if tex != null:
        var sb := StyleBoxTexture.new()
        sb.texture = tex
        sb.set_content_margin_all(get_ui_space("space.md", 16))
        return sb
    # 无贴图时用纯色兜底
    var flat := StyleBoxFlat.new()
    flat.bg_color = get_color("color.accent.warm")
    flat.set_corner_radius_all(get_ui_radius("radius.button", 10))
    flat.set_content_margin_all(get_ui_space("space.md", 16))
    return flat

func _make_panel_stylebox(asset_id: String) -> StyleBox:
    var tex := get_ui_texture(asset_id)
    if tex != null:
        var sb := StyleBoxTexture.new()
        sb.texture = tex
        sb.set_content_margin_all(get_ui_space("space.md", 16))
        return sb
    var flat := StyleBoxFlat.new()
    flat.bg_color = get_color("color.bg.panel")
    flat.border_color = get_color("color.border.soft")
    flat.set_border_width_all(1)
    flat.set_corner_radius_all(get_ui_radius("radius.panel", 14))
    flat.shadow_color = get_color("color.shadow.soft")
    flat.shadow_size = 6
    flat.set_content_margin_all(get_ui_space("space.md", 16))
    return flat

func _make_focus_stylebox() -> StyleBoxFlat:
    var flat := StyleBoxFlat.new()
    flat.bg_color = Color.TRANSPARENT
    flat.border_color = get_color("color.border.focus")
    flat.set_border_width_all(2)
    flat.set_corner_radius_all(get_ui_radius("radius.button", 10))
    return flat
```

## 6. NoticeManager 三通道分离（解决提示重叠）

**文件**：`autoload/notice_manager.gd`（若无则新建）

```gdscript
extends Node

signal notice_requested(channel: String, payload: Dictionary)

enum Channel { NPC, SYSTEM, SCENE }

const CHANNEL_NAMES := {
    Channel.NPC: "npc",
    Channel.SYSTEM: "system",
    Channel.SCENE: "scene",
}

# 每通道独立队列，互不抢占
var _queues := {"npc": [], "system": [], "scene": []}
var _active := {"npc": null, "system": null, "scene": null}

func push(channel: Channel, text: String, speaker: String = "", avatar_key: String = "", duration: float = 3.5) -> void:
    var ch := CHANNEL_NAMES[channel]
    _queues[ch].append({
        "text": text, "speaker": speaker, "avatar_key": avatar_key,
        "duration": duration, "ts": Time.get_ticks_msec(),
    })
    _dispatch(ch)

func _dispatch(ch: String) -> void:
    if _active[ch] != null: return
    if _queues[ch].is_empty(): return
    var payload: Dictionary = _queues[ch].pop_front()
    _active[ch] = payload
    notice_requested.emit(ch, payload)

func dismiss(ch: String) -> void:
    _active[ch] = null
    _dispatch(ch)
```

**HUD 侧**：为每个通道建独立 `PanelContainer`，锚点分离：

| 通道 | 锚点 | 位置 | 尺寸 |
|---|---|---|---|
| `npc` | 底部中央 | `anchor_bottom=1, anchor_left=0.5, anchor_right=0.5`，`offset_top=-140, offset_bottom=-80` | 宽 520 |
| `system` | 顶部中央 | `anchor_top=0, anchor_left=0.5, anchor_right=0.5`，`offset_top=16, offset_bottom=64` | 宽 480 |
| `scene` | 右侧中部 | `anchor_right=1, anchor_top=0.5, anchor_bottom=0.5`，`offset_left=-320, offset_right=-24` | 宽 296 |

三通道物理隔离 → **永不重叠**。

## 11. 风险与缓解

| 风险 | 影响 | 缓解 |
|---|---|---|
| `ui_assets.csv` 贴图未就绪 | 按钮无图 | `_make_button_stylebox` 走 `StyleBoxFlat` 兜底，颜色/圆角/边距全部来自 token，视觉仍统一 |
| `ConfigDB` 未加载新 CSV | PM 返回空 | `get_row` 空时返回 fallback；CSV 加载顺序在 PM `_ready` 之前（现有机制已保证） |
| 三通道 Panel 锚点与 Hotbar 冲突 | 底部遮挡 | NPC 通道 `offset_bottom=-80`，Hotbar 在 `-24`，留 56px 间距 |
| 旧存档引用旧 token 名 | 存档兼容 | 本方案**只新增 token，不删除/重命名**，旧 key 全部保留 |
| `theme_type_variation` 在 Godot 4.7 行为 | 变体不生效 | 已在 §5 用 `theme.set_type_variation` 显式注册；若失效，回退为直接 `add_theme_stylebox_override` |
| `_calendar_panel` 删除后其他脚本引用 | 空引用 | 全局搜索 `_calendar_panel`，改为 `_calendar_chip`；本方案已列在 §7.1 |
| NoticeManager 与现有 `NoticeManager.notice_requested` 签名冲突 | 信号不匹配 | §6 保留 `notice_requested(channel, payload)`，HUD 侧 `show_notice` 改为按 channel 分发；旧调用点统一改走 `push()` |

## 12. 与共享黑板约束的对齐

- ✅ 无任务面板、无数值属性条：StatusBar 只显示钱/时间/天气/日期，非数值条
- ✅ NPC/系统/场景提示分离：§6 三通道物理隔离
- ✅ 温柔治愈、低饱和暖色、柔光：token 全部暖米/暖橙/青绿，`color.glow.warm` + `shadow.soft`
- ✅ 整洁但不冷清：面板留白 `space.md=16`，圆角 14，柔边框 1px
- ✅ 存档兼容：只增不删 token/字段
- ✅ 统一 Theme：`PresentationManager.build_ui_theme()` 单一入口

## 独立方案

# 独立方案：HUD 信息层级重构 + 主题接口改造

> 立场声明：本方案不沿用主方案"在 `hud.gd` 里继续堆 PanelContainer"的路线。核心判断是——**左上角过杂、提示重叠、按钮状态缺失三个问题同源**：`hud.gd` 把"状态显示 / 提示投递 / 模态面板"三种职责塞进同一个 CanvasLayer，且所有视觉常量硬编码在 `_build_interface()` 里。修 UI 之前必须先拆职责，否则改完还会复发。

---

## 0. 问题归因（先定病灶，再开药）

| 症状 | 真实病灶 | 证据 |
|---|---|---|
| 左上角过杂 | `_status_panel` / `_calendar_panel` / `_context_panel` / `_npc_hint_panel` 四个 PanelContainer 各自 `position` 硬编码，无布局容器 | `hud.gd` 中 `_status_panel`、`_calendar_panel`、`_context_panel`、`_npc_hint_panel` 全部独立定位 |
| 提示重叠 | `show_notice()` 与 `_update_npc_hint()` 与 `_update_pointer_panel()` 三条独立通道，无仲裁 | `NoticeManager.notice_requested` 直连 `show_notice`；`_npc_hint_accumulator` 5 秒轮询；`_update_pointer_panel` 每帧 |
| 按钮状态缺失 | `_make_menu_button` 只设 `text` + `custom_minimum_size`，无 `theme` 无 `StyleBox` 无 hover/pressed/disabled 定义 | `main_menu.gd` 第 88-95 行 |

**结论**：需要引入 **HUD 分区（Zone）+ 提示仲裁器（NoticeArbiter）+ 主题令牌（ThemeTokens）** 三层，而不是继续加 Panel。

## 1. 架构改造（与主方案的关键分歧）

### 1.1 分区模型：把左上角拆成 3 个 Zone，用容器而非坐标

```
HUD (CanvasLayer, layer=20)
└── Root (Control, FULL_RECT, MOUSE_IGNORE)
    ├── ZoneTopLeft (VBoxContainer, anchor=TL, offset=(16,16))
    │   ├── StatusBar      ← 金钱/时间/天气 合并为一行 chip
    │   └── ContextChip    ← 当前场景/可交互对象（原 _context_panel）
    ├── ZoneTopRight (VBoxContainer, anchor=TR, offset=(-16,16))
    │   └── CalendarChip   ← 日期/季节（原 _calendar_panel）
    ├── ZoneBottomCenter (VBoxContainer, anchor=BC, offset=(0,-16))
    │   ├── NoticeStack    ← 提示仲裁器输出（最多 1 条可见）
    │   └── HotbarPanel    ← 原 _hotbar_panel
    ├── ZoneBottomRight (VBoxContainer, anchor=BR, offset=(-16,-16))
    │   └── PointerHint    ← 原 _pointer_panel
    └── ModalLayer (Control, FULL_RECT)  ← 所有 ModalState 面板挂这里
```

**关键点**：
- 左上角从 4 个 Panel 降为 **2 个 chip**（StatusBar 合并金钱/时间/天气；ContextChip 保留场景上下文）。
- `_npc_hint_panel` **不再独立存在**，并入 NoticeStack，由仲裁器决定优先级。
- 所有位置通过 `anchor + offset` 表达，不再写 `position = Vector2(...)`。

### 1.2 提示仲裁器：单一投递通道

新增 `autoload/notice_arbiter.gd`，替代 `NoticeManager.notice_requested` 直连 HUD 的做法。

```gdscript
# autoload/notice_arbiter.gd
extends Node

enum Priority { AMBIENT = 0, NPC_HINT = 1, SYSTEM = 2, CRITICAL = 3 }

signal notice_pushed(notice: Dictionary)
signal notice_cleared(notice_id: String)

const MAX_VISIBLE := 1
const QUEUE_CAP := 8

var _active: Dictionary = {}          # 当前可见
var _queue: Array[Dictionary] = []    # 等待队列
var _cooldowns: Dictionary = {}       # notice_id -> expire_time

func push(notice_id: String, text: String, priority: int = Priority.SYSTEM,
        duration: float = 3.0, speaker: String = "", avatar_key: String = "",
        source_kind: String = "system", cooldown: float = 0.0) -> void:
    var now := Time.get_ticks_msec() / 1000.0
    if cooldown > 0.0 and _cooldowns.get(notice_id, 0.0) > now:
        return
    _cooldowns[notice_id] = now + cooldown
    var entry := {
        "id": notice_id, "text": text, "priority": priority,
        "duration": duration, "speaker": speaker, "avatar_key": avatar_key,
        "source_kind": source_kind, "pushed_at": now,
    }
    if _active.is_empty():
        _activate(entry)
    elif priority > int(_active.priority):
        _queue.push_front(_active)   # 抢占：当前降级回队列
        _activate(entry)
    else:
        if _queue.size() >= QUEUE_CAP:
            _queue.pop_back()
        _queue.push_back(entry)

func _activate(entry: Dictionary) -> void:
    _active = entry
    notice_pushed.emit(entry)

func tick(delta: float) -> void:
    if _active.is_empty():
        if not _queue.is_empty():
            _activate(_queue.pop_front())
        return
    _active["duration"] -= delta
    if _active["duration"] <= 0.0:
        var finished_id: String = _active["id"]
        _active = {}
        notice_cleared.emit(finished_id)
        if not _queue.is_empty():
            _activate(_queue.pop_front())
```

**HUD 侧改动**：`_process` 里调用 `NoticeArbiter.tick(delta)`，`NoticeStack` 只监听 `notice_pushed` / `notice_cleared` 两个信号。`_npc_hint_accumulator` 逻辑改为 `NoticeArbiter.push("npc_hint_%s" % npc_id, ..., Priority.NPC_HINT, 4.0, cooldown=8.0)`。

### 1.3 按钮状态：Theme 驱动，不再逐按钮 override

`main_menu.gd` 的 `_make_menu_button` 删除所有 `add_theme_*_override`，改为在 `_root.theme` 上挂 `ThemeTokens.build_theme()`。按钮状态由 `theme.set_stylebox("normal"/"hover"/"pressed"/"disabled"/"focus", "Button", ...)` 统一提供。

## 5. PresentationManager API 改造

在现有 `autoload/presentation_manager.gd` 追加（不破坏现有方法）：

```gdscript
# --- Theme tokens ---
func get_int(token: String, fallback: int = 0) -> int:
    return int(get_token(token, str(fallback)))

func get_stylebox(token: String) -> StyleBox:
    var path := "res://assets/art/ui/%s.png" % token.trim_prefix("ui.")
    var tex := _load_texture(path)
    if tex == null:
        return null
    var sb := StyleBoxTexture.new()
    sb.texture = tex
    sb.texture_margin_left = 12
    sb.texture_margin_right = 12
    sb.texture_margin_top = 12
    sb.texture_margin_bottom = 12
    return sb

# --- UI assets ---
func get_ui_texture(asset_key: String) -> Texture2D:
    var path := "res://assets/art/ui/%s.png" % asset_key.trim_prefix("ui.")
    return _load_texture(path)

# --- Text styles ---
func apply_text_style(label: Label, style_id: String) -> void:
    var row := ConfigDB.get_row("ui_text_styles", style_id)
    if row.is_empty():
        return
    var font_names := get_font_names(str(row.get("font_role", "font.body")))
    var font := SystemFont.new()
    font.font_names = font_names
    label.add_theme_font_override("font", font)
    label.add_theme_font_size_override("font_size", get_int(str(row.get("size_token", "font.size.body")), 18))
    label.add_theme_color_override("font_color", get_color(str(row.get("color_token", "color.text.primary"))))
    var outline := int(row.get("outline_size", 0))
    if outline > 0:
        label.add_theme_constant_override("outline_size", outline)
        label.add_theme_color_override("font_outline_color", get_color(str(row.get("outline_color", "color.text.primary"))))
    label.add_theme_constant_override("line_spacing", int(row.get("line_spacing", 0)))

# --- Theme builder (供 MainMenu / HUD 共用) ---
func build_theme() -> Theme:
    var theme := Theme.new()
    var font := SystemFont.new()
    font.font_names = get_font_names("font.body")
    theme.default_font = font
    theme.default_font_size = get_int("font.size.body", 18)

    for state in ["normal", "hover", "pressed", "disabled", "focus"]:
        var sb := get_stylebox("ui.button.%s" % state)
        if sb != null:
            theme.set_stylebox(state, "Button", sb)
    theme.set_color("font_color", "Button", get_color("color.text.primary"))
    theme.set_color("font_hover_color", "Button", get_color("color.text.primary"))
    theme.set_color("font_pressed_color", "Button", get_color("color.text.primary"))
    theme.set_color("font_disabled_color", "Button", get_color("color.text.muted"))
    theme.set_color("font_focus_color", "Button", get_color("color.text.primary"))

    var panel_sb := get_stylebox("ui.panel.bg")
    if panel_sb != null:
        theme.set_stylebox("panel", "PanelContainer", panel_sb)
    return theme
```

**兼容性**：现有 `get_token` / `get_color` / `get_font_names` 签名不变，新增方法纯追加。

# 批判与返工要求

---

## 一、主方案（HUD 信息层级与主题接口重构方案）的致命缺陷

### M1. `build_ui_theme()` 里有一行**死代码**，且 Button 四态注册逻辑自相矛盾

```gdscript
theme.set_stylebox("normal", "Button", normal) if variant == "primary" else null
```

- 这是**表达式语句**，Godot 4 会报 `Standalone expression` 警告，且 `else null` 分支无副作用 → 完全死代码。
- 更严重：`primary` 变体把 normal 塞给基类 `Button`，`ghost` 变体不塞。结果**所有未指定 variation 的 Button 会继承 primary 的 normal**，但 hover/pressed/disabled 又没设到基类 → 未指定 variation 的按钮出现"常态是 primary、hover 无样式"的**半残状态**。
- 同时 `theme.set_type_variation("PrimaryButton", "Button")` 在 `Button` 基类已被污染的情况下，变体继承链不可预测。

**返工要求**：
1. 删除该三元表达式行。
2. 明确二选一：要么**不给基类 `Button` 设任何 stylebox**，所有按钮必须显式 `theme_type_variation`；要么**给基类设完整四态**（用 ghost 或 neutral 作为默认）。禁止"基类只设 normal"。

## 二、独立方案的致命缺陷

### I1. `NoticeArbiter` 的**抢占逻辑会丢消息**

```gdscript

    _queue.push_front(_active)   # 抢占：当前降级回队列
    _activate(entry)
```

- 被抢占的 `_active` 的 `duration` **已经被 tick 扣过一部分**，push_front 回队列后，下次激活时 `duration` 是**剩余值**而非原始值 → 显示时间变短，甚至为负立即消失。
- 更严重：`_queue.push_front` 后队列顺序被打乱，若连续抢占，队列头部堆积多条"被降级"的旧消息，**新消息反而排在后面**。

**返工要求**：
1. 被抢占时**重置 duration 为原始值**（entry 里存 `original_duration`）。
2. 明确队列策略：抢占回队列是 push_front 还是 push_back？建议 push_back + 记录 `preempted_at`，避免旧消息永远插队。
3. 补断言：push SYSTEM(duration=3) → 0.5s 后 push CRITICAL → CRITICAL 显示 → CRITICAL 结束后 SYSTEM **重新显示满 3 秒**（或明确降级为剩余 2.5s，但要在文档写清）。

### I2. `tick(delta)` 依赖 HUD `_process` 调用，但 **HUD 可能被 pause / 隐藏**

## 一、决策（裁决表）

| 争议点 | 主方案 | 独立方案 | **最终裁决** | 理由 |
|---|---|---|---|---|
| 左上角布局 | 单 PanelContainer 内嵌 HBox | Zone 容器 + chip 组件 | **Zone + chip** | 容器驱动，换分辨率不破；批判 I5 要求禁硬编码 offset |
| 提示通道 | NoticeManager 三通道物理隔离 | NoticeArbiter 单通道 + 优先级抢占 | **NoticeArbiter 单通道** | 批判 M4 证明"物理隔离"是伪保证；单通道 + 优先级可仲裁 |
| 按钮状态 | Theme type_variation 四态 | Theme 统一 5 态 | **Theme 统一 5 态（含 focus）** | 批判 M1 指出基类污染；批判 C1 要求 focus 导航 |
| 主题构建 | `build_ui_theme()` 每次新建 | `build_theme()` 每次新建 | **`build_theme()` + 缓存 + `invalidate_theme()`** | 批判 C4 要求性能预算 |
| 资产接入 | CSV 表 + fallback 链 | 路径拼接 | **CSV 表 + 缓存 + 环检测** | 批判 I3 证明路径拼接必 404；批判 M2 要求防环 |
| 字体列表 | CSV 逗号分隔 | 未明确 | **分号分隔 + split** | 批判 M7 证明逗号冲突 |
| money 刷新 | 塞进 `_update_clock` | 未明确 | **信号驱动，独立于定时器** | 批判 M6 要求同帧刷新 |
| 提示计时 | HUD 侧无 dismiss | HUD `_process` 驱动 tick | **Arbiter 自驱 + pause 冻结** | 批判 M5/I2 要求不依赖 HUD |

**核心原则**：
1. **单一数据源**：所有视觉常量走 `theme_tokens.csv`，所有文本样式走 `ui_text_styles.csv`，所有贴图走 `ui_assets.csv`。代码中**禁止出现 `Color("#...")`、`Vector2(16,16)` 定位、`font_size = 22` 字面量**。
2. **单一投递通道**：所有提示走 `NoticeArbiter.push()`，HUD 只渲染。
3. **单一主题入口**：`PresentationManager.build_theme()` 缓存构建，MainMenu / HUD 共用。
4. **可回退**：feature flag `ui.theme.new_pipeline`，false 时走旧路径。

## 四、PresentationManager API（修复 M1/M2/M3/M7/C4）

```gdscript
# ============ 缓存 ============
var _texture_cache: Dictionary = {}
var _theme_cache: Theme = null

# ============ 基础读取 ============
func get_int(token: String, fallback: int = 0) -> int:
    var v := get_token(token, str(fallback))
    return int(v) if v.is_valid_int() else fallback

func get_font_names(role: String) -> PackedStringArray:
    var raw := get_token(role, "")
    if raw.is_empty():
        return PackedStringArray(["sans-serif"])
    # 分号分隔（修复 M7）
    var parts := raw.split(";")
    var out := PackedStringArray()
    for p in parts:
        var s := p.strip_edges()
        if not s.is_empty():
            out.append(s)
    return out

# ============ 文本样式 ============
func get_text_style(style_id: String) -> Dictionary:
    return ConfigDB.get_row("ui_text_styles", style_id)

func apply_text_style(label: Label, style_id: String) -> void:
    var s := get_text_style(style_id)
    if s.is_empty():
        push_warning("[PM] text style not found: %s" % style_id)
        return
    var font := SystemFont.new()
    font.font_names = get_font_names(str(s.get("font_role", "font.body")))
    label.add_theme_font_override("font", font)
    label.add_theme_font_size_override("font_size",
        get_int(str(s.get("size_token", "font.size.body")), 18))
    label.add_theme_color_override("font_color",
        get_color(str(s.get("color_token", "color.text.primary"))))
    # outline 分支（修复 M3：CSV 中 title.* 已配 outline）
    var outline_size := int(s.get("outline_size", 0))
    var outline_token := str(s.get("outline_color_token", ""))
    if outline_size > 0 and not outline_token.is_empty():
        label.add_theme_constant_override("outline_size", outline_size)
        label.add_theme_color_override("font_outline_color", get_color(outline_token))
    label.add_theme_constant_override("line_spacing", int(s.get("line_spacing", 0)))
    # 文本溢出（修复 C3）
    label.text_overrun_behavior = TextServer.OVERRUN_TRIM_ELLIPSIS
    label.clip_text = true

func get_ui_texture(asset_id: String) -> Texture2D:
    return _get_ui_texture_recursive(asset_id, {}, 0)

func _get_ui_texture_recursive(asset_id: String, visited: Dictionary, depth: int) -> Texture2D:
    if depth > 4:
        push_warning("[PM] texture fallback depth exceeded: %s" % asset_id)
        return null
    if visited.has(asset_id):
        push_warning("[PM] texture fallback cycle: %s" % asset_id)
        return null
    if _texture_cache.has(asset_id):
        return _texture_cache[asset_id]
    visited[asset_id] = true
    var row := ConfigDB.get_row("ui_assets", asset_id)
    if row.is_empty():
        _texture_cache[asset_id] = null
        return null
    var path := str(row.get("path", ""))
    var tex := _load_texture(path)
    if tex == null:
        var fb := str(row.get("fallback_asset_id", ""))
        if not fb.is_empty():
            tex = _get_ui_texture_recursive(fb, visited, depth + 1)
    _texture_cache[asset_id] = tex
    return tex

func invalidate_texture_cache() -> void:
    _texture_cache.clear()

func build_theme() -> Theme:
    if _theme_cache != null:
        return _theme_cache
    var theme := Theme.new()
    var body_font := SystemFont.new()
    body_font.font_names = get_font_names("font.body")
    theme.default_font = body_font
    theme.default_font_size = get_int("font.size.body", 18)

    # --- Label 全局 ---
    theme.set_color("font_color", "Label", get_color("color.text.primary"))
    theme.set_font_size("font_size", "Label", get_int("font.size.body", 18))

    # --- Button 基类：完整 5 态（用 ghost 作默认，修复 M1） ---
    _apply_button_variation(theme, "Button", "ghost")
    # --- 显式变体 ---
    theme.set_type_variation("PrimaryButton", "Button")
    _apply_button_variation(theme, "PrimaryButton", "primary")
    theme.set_type_variation("GhostButton", "Button")
    _apply_button_variation(theme, "GhostButton", "ghost")

    # --- PanelContainer ---
    theme.set_stylebox("panel", "PanelContainer", _make_panel_stylebox("panel.paper"))
    theme.set_type_variation("SoftPanel", "PanelContainer")
    theme.set_stylebox("panel", "SoftPanel", _make_panel_stylebox("panel.paper_soft"))
```

## 五、NoticeArbiter（修复 M5/I1/I2/I7）

**文件**：`autoload/notice_arbiter.gd`

```gdscript
extends Node

signal notice_pushed(entry: Dictionary)
signal notice_cleared(notice_id: String)

enum Priority { AMBIENT = 0, NPC_HINT = 1, SYSTEM = 2, CRITICAL = 3 }

const MAX_QUEUE := 8
const MAX_VISIBLE := 1

var _active: Dictionary = {}
var _queue: Array[Dictionary] = []
var _cooldowns: Dictionary = {}   # notice_id -> expire_ticks_msec

func _ready() -> void:
    # 自驱，不依赖 HUD（修复 I2）
    process_mode = Node.PROCESS_MODE_ALWAYS
    set_process(true)

func _process(delta: float) -> void:
    # pause 时冻结 duration（修复 I2）
    if get_tree().paused:
        return
    _tick(delta)

func push(notice_id: String, text: String, priority: int = Priority.SYSTEM,
        duration: float = 3.0, speaker: String = "", avatar_key: String = "",
        source_kind: String = "system", cooldown: float = 0.0) -> void:
    var now := Time.get_ticks_msec()
    if cooldown > 0.0:
        var expire: int = _cooldowns.get(notice_id, 0)
        if expire > now:
            return
        _cooldowns[notice_id] = now + int(cooldown * 1000.0)
    var entry := {
        "id": notice_id, "text": text, "priority": priority,
        "duration": duration, "original_duration": duration,  # 修复 I1
        "speaker": speaker, "avatar_key": avatar_key,
        "source_kind": source_kind,
    }
    if _active.is_empty():
        _activate(entry)
    elif priority > int(_active.priority):
        # 抢占：当前降级回队列，重置 duration（修复 I1）
        var preempted: Dictionary = _active.duplicate()
        preempted["duration"] = preempted["original_duration"]
        _enqueue(preempted)
        _activate(entry)
    else:
        _enqueue(entry)

func _enqueue(entry: Dictionary) -> void:
    if _queue.size() >= MAX_QUEUE:
        _queue.pop_back()
    _queue.push_back(entry)   # 修复 I1：push_back，不插队

func _activate(entry: Dictionary) -> void:
    _active = entry
    notice_pushed.emit(entry)

func _tick(delta: float) -> void:
    if _active.is_empty():
        if not _queue.is_empty():
            _activate(_queue.pop_front())
        return
    _active["duration"] -= delta
    if _active["duration"] <= 0.0:
        var finished_id: String = _active["id"]
        _active = {}
        notice_cleared.emit(finished_id)
        if not _queue.is_empty():
            _activate(_queue.pop_front())

func clear_all() -> void:
    _active = {}
    _queue.clear()

func reset_cooldowns() -> void:
    # 会话级冷却，重启归零（修复 I7：文档明确）
    _cooldowns.clear()
```

**关键修复对照**：
- M5：`_tick` 内自动 dismiss，HUD 无需调 `dismiss`。
- I1：`original_duration` 保底；`push_back` 不插队。
- I2：`process_mode = PROCESS_MODE_ALWAYS` + `get_tree().paused` 冻结。
- I7：`reset_cooldowns()` 明确会话级语义。
