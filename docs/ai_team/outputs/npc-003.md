## 1. 决策（逐条裁决）

| # | 争议点 | 最终裁决 | 理由 |
|---|---|---|---|
| D1 | 生日事件载体 | **新建 `data/npc_birthdays.csv`**，不塞进 `npcs.csv` | `npcs.csv` 已有 `birthday_day` 列，但生日**事件内容**（台词/奖励/解锁）是独立维度，塞进去会让单行 20+ 列 |
| D2 | 生日触发时机 | **当天首次进入该 NPC 所在场景时触发**，非 `day_started` 立即弹 | 玩家可能一整天不去工厂，`day_started` 弹王师傅生日会错位 |
| D3 | 生日奖励 | **固定小礼物 + 一次性 affinity +2**，无金钱 | 温柔治愈；金钱奖励会让生日变刷钱 |
| D4 | 生日台词通道 | **NPC 提示**（`source_kind=npc`），speaker = NPC 名 | 生日是 NPC 说的话，不是系统播报 |
| D5 | 节日限时 NPC | **复用 `lan`（阿榕）**，不新增。`lan` 的 `birthday_day=359` 恰为年末，其"节日摊"已覆盖 | 现有 `lan` 定位就是"节日摊主，每年换地方" |
| D6 | 节日限时出现 | **`condition` 列统一为 `always / weekend / festival:N / season:X / event:Y`**（与 npc-001 一致） | 单一语法 |
| D7 | 长期彩蛋载体 | **新建 `data/npc_eggs.csv`**，与 `npc_stories.csv` 并列 | stories 是"好感度解锁的剧情"，eggs 是"条件触发的隐藏反应"，语义不同 |
| D8 | 彩蛋触发 | **纯条件驱动**（flag + 时段 + 天气 + 场景），无概率 | 可测试；概率彩蛋玩家永远不知道有没有 |
| D9 | 彩蛋提示通道 | **NPC 提示**，但 `reliability_hint` 可空 | 彩蛋是 NPC 的额外反应 |
| D10 | 性格化提示 | **`npc_dialogue.csv` 新增 `tone` 列**（`warm / dry / brisk / quiet / playful`），选择器按 tone 过滤 | 现有 `personality` 列是自由文本，不可机读 |
| D11 | 可靠度 | **`npc_dialogue.csv` 新增 `reliability` 列**（`high / medium / low`，默认 `high`） | npc-002 已裁决三档枚举 |
| D12 | 生日跨年 | `birthday_greeted` 键改为 `npc_id_year` | 否则第二年不再触发 |
| D13 | 生日与节日同日 | **生日让位**，当天只播节日，生日顺延至次日 | 避免提示抢占（npc-002 D7） |
| D14 | 限时 NPC 离场 | `lan` 在非节日期间**不实例化**，不是隐藏 | 避免空气墙/交互空点 |
| D15 | 彩蛋奖励 | **只给 item 或 affinity，不给 money** | 彩蛋是惊喜不是收入 |

### 3.1 修改 `autoload/relationship_manager.gd`

**新增**：
```gdscript
const BIRTHDAY_CSV := "npc_birthdays"
const EGG_CSV := "npc_eggs"

var birthday_met: Dictionary = {}   # key: "npc_id_year" -> true
var egg_met: Dictionary = {}        # key: egg_id -> true

func get_birthday_npcs(day_number: int) -> Array:
    var result: Array = []
    var doy := CalendarManager.get_day_of_year(day_number)
    for row in ConfigDB.get_rows(BIRTHDAY_CSV):
        if int(row.get("day_of_year", -1)) == doy:
            result.append(str(row.get("npc_id", "")))
    return result

func get_birthday_row(npc_id: String) -> Dictionary:
    for row in ConfigDB.get_rows(BIRTHDAY_CSV):
        if str(row.get("npc_id", "")) == npc_id:
            return row
    return {}

func try_trigger_birthday(npc_id: String, scene_key: String) -> bool:
    var row := get_birthday_row(npc_id)
    if row.is_empty():
        return false
    if str(row.get("trigger_scene", "")) != scene_key:
        return false
    var year := CalendarManager.get_year()
    var key := "%s_%d" % [npc_id, year]
    if birthday_met.get(key, false):
        return false
    # 与节日同日则让位
    if CalendarManager.is_festival():
        return false
    var aff := int(affinity.get(npc_id, 0))
    var line := str(row.get("npc_line_low", "")) if aff < 4 else str(row.get("npc_line", ""))
    NoticeManager.show_npc_message(line, str(row.get("npc_id", "")))
    _apply_reward(row)
    birthday_met[key] = true
    var flag := str(row.get("unlock_flag", ""))
    if not flag.is_empty():
        GameState.set_flag(flag, true)
    return true

func try_trigger_egg(npc_id: String, scene_key: String) -> bool:
    for row in ConfigDB.get_rows(EGG_CSV):
        if str(row.get("npc_id", "")) != npc_id:
            continue
        var egg_id := str(row.get("egg_id", ""))
        if bool(row.get("once", false)) and egg_met.get(egg_id, false):
            continue
        if not _egg_condition_ok(row, scene_key):
            continue
        NoticeManager.show_npc_message(str(row.get("line", "")), npc_id)
        _apply_reward(row)
        egg_met[egg_id] = true
        var flag := str(row.get("unlock_flag", ""))
        if not flag.is_empty():
            GameState.set_flag(flag, true)
        return true
    return false

func _egg_condition_ok(row: Dictionary, scene_key: String) -> bool:
    var cond := str(row.get("condition", "always"))
    if cond == "weekend" and not TimeSystem.is_weekend():
        return false
    if cond.begins_with("festival:") and not CalendarManager.is_festival():
        return false
    if cond.begins_with("season:") and CalendarManager.get_season_id() != cond.split(":")[1]:
        return false
    var weather := str(row.get("weather", ""))
    if not weather.is_empty() and WeatherSystem.current_weather_id != weather:
        return false
    var period := str(row.get("period", ""))
    if not period.is_empty() and TimeSystem.get_period() != period:
        return false
    var flag := str(row.get("required_flag", ""))
    if not flag.is_empty() and not GameState.has_flag(flag):
        return false
    if int(row.get("min_affinity", 0)) > int(affinity.get(str(row.get("npc_id", "")), 0)):
        return false
    return true

func _apply_reward(row: Dictionary) -> void:
    var rtype := str(row.get("reward_type", ""))
    if rtype == "item":
        InventoryManager.add_item(str(row.get("reward_item", "")), int(row.get("reward_value", 1)))
    elif rtype == "affinity":
        var npc_id := str(row.get("npc_id", ""))
        affinity[npc_id] = int(affinity.get(npc_id, 0)) + int(row.get("reward_value", 1))
        changed.emit()
```

## 4. 触发条件（唯一语法）

**统一语法**：`condition = "kind:value"`，五种 kind：

| kind | 语义 | 示例 |
|---|---|---|
| `always` | 常驻 | `always` |
| `weekend` | 周末 | `weekend` |
| `festival:<id>` | 节日当天 | `festival:lantern` |
| `season:<id>` | 季节 | `season:summer` |
| `event:<id>` | 事件周 | `event:city_festival_week` |

**生日是特殊 kind**：`birthday:<npc_id>`，由 `CalendarManager` 在 `day_started` 时计算。

**触发优先级**（高 → 低）：
1. `birthday:*`（一年一次）
2. `festival:*`（节日当天）
3. `event:*`（事件周）
4. `season:*`（季节）
5. `weekend`
6. `always`

**冲突裁决**：同一天多个事件触发时，**场景实体状态叠加**（生日桌子 + 节日灯笼可以同时存在），但 **NPC 对话只取最高优先级**。

## 二、主方案的具体漏洞

### M1. `get_birthday_npcs` 签名不一致（死代码风险）
- §3.1 定义：`func get_birthday_npcs(day_number: int) -> Array`

### M2. `try_trigger_birthday` 的 `trigger_scene` 检查与 D2 矛盾
- D2 说"当天首次进入该 NPC 所在场景时触发"。
- §3.1 代码：`if str(row.get("trigger_scene", "")) != scene_key: return false`
- **问题**：如果 NPC 在多个场景出现（如 `lin` 在 `store` 和 `street` 都可能出现），`trigger_scene` 只写一个场景，玩家在另一个场景遇到 `lin` 就**永远不触发**。
- **返工要求**：确认每位 NPC 是否只在一个场景出现。若不是，`trigger_scene` 必须支持多值（逗号分隔）或改为"NPC 当前所在场景"动态判断。

### M3. `_egg_condition_ok` 的 `festival:` 检查是死代码
- §3.1：`if cond.begins_with("festival:") and not CalendarManager.is_festival(): return false`
- **问题**：`is_festival()` 只返回"今天是不是节日"，**不检查是不是 `festival:1` 这个特定节日**。`lan_egg_festival` 的 `condition=festival:1`，但代码只检查"是不是节日"，不检查"是不是 1 号节日"。
- **返工要求**：改为 `CalendarManager.get_festival_id() == cond.split(":")[1]`，或明确 `is_festival()` 的语义。

### M4. `_egg_condition_ok` 的 `season:` 检查用 `!=` 比较字符串

```gdscript

func try_show_birthday_prop(npc_id: String, scene_key: String) -> bool:
    var row := get_birthday_row(npc_id)
    if row.is_empty():
        return false
    var scenes := str(row.get("trigger_scene", "")).split(",")
    if not scene_key in scenes:
        return false
    var year := CalendarManager.get_year()
    var day := CalendarManager.get_day_of_year(CalendarManager.current_day)
    var key := "%s_%d_%d" % [npc_id, year, day]
    if birthday_scene_shown.get(key, false):
        return false
    birthday_scene_shown[key] = true
    return true  # 调用方据此实例化 scene_prop_id

func try_trigger_birthday(npc_id: String, scene_key: String) -> bool:

    NoticeManager.show_npc_message(line, npc_id)

    birthday_met[key] = true
    var flag := str(row.get("unlock_flag", ""))
    if not flag.is_empty():
        GameState.set_flag(flag, true)
    return true

func try_trigger_egg(npc_id: String, scene_key: String) -> bool:
    for row in ConfigDB.get_rows(EGG_CSV):
        if str(row.get("npc_id", "")) != npc_id:
            continue
        var egg_id := str(row.get("egg_id", ""))
        if bool(row.get("once", false)) and egg_met.get(egg_id, false):
            continue
        if not _egg_condition_ok(row, scene_key):
            continue
        NoticeManager.show_npc_message(str(row.get("line", "")), npc_id)
        _apply_reward(row, npc_id)
        egg_met[egg_id] = true
        var flag := str(row.get("unlock_flag", ""))
        if not flag.is_empty() and not GameState.has_flag(flag):
            GameState.set_flag(flag, true)
        return true
    return false

func _egg_condition_ok(row: Dictionary, scene_key: String) -> bool:
    var scene := str(row.get("scene", ""))
    if not scene.is_empty() and scene != scene_key:
        return false
    var cond := str(row.get("condition", "always"))
    if cond == "weekend" and not TimeSystem.is_weekend():
        return false
    if cond.begins_with("festival:"):
        var fid := cond.split(":")[1]
        if str(CalendarManager.get_festival_id()) != fid:
            return false
    if cond.begins_with("season:"):
        var sid := cond.split(":")[1]
        if str(CalendarManager.get_season_id()) != sid:
            return false
    if cond.begins_with("event:"):
        var eid := cond.split(":")[1]
        if not CalendarManager.is_event_active(eid):
            return false
    var weather := str(row.get("weather", ""))
    if not weather.is_empty() and WeatherSystem.current_weather_id != weather:
        return false
    var period := str(row.get("period", ""))
    if not period.is_empty() and TimeSystem.get_period() != period:
        return false
    var flag := str(row.get("required_flag", ""))
    if not flag.is_empty() and not GameState.has_flag(flag):
        return false
    if int(row.get("min_affinity", 0)) > int(affinity.get(str(row.get("npc_id", "")), 0)):
        return false
    return true

func _apply_reward(row: Dictionary, npc_id: String) -> void:
    var rtype := str(row.get("reward_type", ""))
    if rtype == "item":
        var item := str(row.get("reward_item", ""))
        if item.is_empty():
            push_error("reward_type=item but reward_item empty: " + str(row))
            return
        InventoryManager.add_item(item, int(row.get("reward_value", 1)))
    elif rtype == "affinity":
        affinity[npc_id] = int(affinity.get(npc_id, 0)) + int(row.get("reward_value", 1))
        changed.emit()
    elif rtype.is_empty():
        pass  # 无奖励
    else:
        push_error("unknown reward_type: " + rtype)
```
